# Профиль Средний Голосов

Как считается средний рейтинг


// Узел: post26112026/ratings/{profileId}
// { fio, src, gender, score, wins, losses, votes }

// Голос: победитель +5 и wins+1, проигравший −5 и losses+1
function bump(profile, delta, isWin) {
  const ref = RATINGS.child(profile.id);
  return ref.transaction(cur => {
    const d = cur || { fio: profile.fio, src: profile.src,
                       gender: profile.gender,
                       score: 0, wins: 0, losses: 0, votes: 0 };
    d.score  += delta;         // +5 или −5
    d.votes  += 1;             // всего голосований
    if (isWin) d.wins  += 1;   // побед
    else       d.losses += 1;  // поражений
    return d;
  });
}

// Средний рейтинг в формате 4.8:
// 5 × победы / голоса, округление до 1 знака
function avgRating(profile) {
  if (!profile.votes) return 0;
  return +(5 * profile.wins / profile.votes).toFixed(1);
}
