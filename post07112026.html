1. Что делает server.js

server.js — это HTTP-сервер на встроенном модуле node:http, который реализует REST-API по образцу JSONPlaceholder. Он предоставляет три ресурса — users, posts, comments — и хранит данные в локальном файле SQLite (api.db).

Ключевые особенности реализации:

  • Драйвер node:sqlite (встроенный в Node.js с версии 22.5) — не требует npm install;
  • Файл базы создаётся рядом со скриптом и сохраняется между перезапусками;
  • Включены WAL (журналирование) и foreign_keys (каскадное удаление);
  • Поддерживаются методы GET, POST, PUT, PATCH, DELETE и OPTIONS (CORS preflight);
  • Один файл, ноль внешних зависимостей.

2. Инициализация базы данных

Первые строки файла подключают два встроенных модуля Node.js и открывают соединение с базой. DatabaseSync — синхронный драйвер: каждая операция сразу возвращает результат, без await. 

import { createServer } from 'node:http';
import { DatabaseSync } from 'node:sqlite';

const db = new DatabaseSync('api.db');
db.exec('PRAGMA journal_mode = WAL;');
db.exec('PRAGMA foreign_keys = ON;');

Что означают два PRAGMA:

  • journal_mode = WAL — режим Write-Ahead Logging. Быстрее, чем журнал по умолчанию, и позволяет читать базу во время записи;
  • foreign_keys = ON — включает проверку внешних ключей. Без этой строки ON DELETE CASCADE в схеме работать не будет.

3. Схема таблиц

Все три таблицы создаются одним вызовом db.exec() с конструкцией CREATE TABLE IF NOT EXISTS — значит, повторный запуск сервера ничего не сломает. 

CREATE TABLE IF NOT EXISTS users (
  id       INTEGER PRIMARY KEY AUTOINCREMENT,
  name     TEXT NOT NULL,
  username TEXT NOT NULL,
  email    TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS posts (
  id     INTEGER PRIMARY KEY AUTOINCREMENT,
  userId INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  title  TEXT NOT NULL,
  body   TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS comments (
  id     INTEGER PRIMARY KEY AUTOINCREMENT,
  postId INTEGER NOT NULL REFERENCES posts(id) ON DELETE CASCADE,
  name   TEXT NOT NULL,
  email  TEXT NOT NULL,
  body   TEXT NOT NULL
);

Связи между таблицами:

  • posts.userId → users.id — один пользователь пишет много постов;
  • comments.postId → posts.id — у одного поста много комментариев;
  • ON DELETE CASCADE — удаление пользователя автоматически уносит его посты и комментарии к ним.

4. Автозаполнение (seeding)

Если таблица users пуста, скрипт заполняет её тестовыми данными — 10 пользователей, 100 постов, 300 комментариев. Это даёт «из коробки» поведение, похожее на JSONPlaceholder. 

if (db.prepare('SELECT COUNT(*) AS c FROM users').get().c === 0) {
  const insUser = db.prepare('INSERT INTO users (name, username, email) VALUES (?, ?, ?)');
  const insPost = db.prepare('INSERT INTO posts (userId, title, body) VALUES (?, ?, ?)');
  const insCmnt = db.prepare('INSERT INTO comments (postId, name, email, body) VALUES (?, ?, ?, ?)');

  db.exec('BEGIN');
  try {
    for (let i = 1; i <= 10; i++)
      insUser.run(`User ${i}`, `user${i}`, `user${i}@example.com`);

    for (let i = 1; i <= 100; i++)
      insPost.run(((i - 1) % 10) + 1, `Post ${i} title`, `Post ${i} body`);

    for (let i = 1; i <= 300; i++)
      insCmnt.run(((i - 1) % 100) + 1, `Commenter ${i}`, `c${i}@example.com`, `Comment ${i} body`);

    db.exec('COMMIT');
    console.log('DB seeded');
  } catch (e) {
    db.exec('ROLLBACK');
    throw e;
  }
}

Вставка обёрнута в транзакцию (BEGIN … COMMIT): если что-то упадёт в середине, все изменения откатятся через ROLLBACK. Это заметно быстрее, чем 410 отдельных INSERT, каждый из которых сам по себе — транзакция.

5. Подготовленные SQL-запросы

Все запросы, которые нужны обработчикам, готовятся один раз при старте и складываются в объект stmt. SQLite компилирует такой запрос при подготовке и потом выполняет его повторно — быстрее и безопаснее (параметры передаются через ?, а не склейкой строк — защита от SQL-инъекций). 

const stmt = {
  users: {
    all:    db.prepare('SELECT * FROM users'),
    byId:   db.prepare('SELECT * FROM users WHERE id = ?'),
    insert: db.prepare('INSERT INTO users (name, username, email) VALUES (?, ?, ?)'),
    update: db.prepare(`UPDATE users
      SET name     = COALESCE(?, name),
          username = COALESCE(?, username),
          email    = COALESCE(?, email)
      WHERE id = ?`),
    del:    db.prepare('DELETE FROM users WHERE id = ?'),
  },
  posts: {
    all:    db.prepare('SELECT * FROM posts'),
    byId:   db.prepare('SELECT * FROM posts WHERE id = ?'),
    byUser: db.prepare('SELECT * FROM posts WHERE userId = ?'),
    insert: db.prepare('INSERT INTO posts (userId, title, body) VALUES (?, ?, ?)'),
    update: db.prepare(`UPDATE posts
      SET userId = COALESCE(?, userId),
          title  = COALESCE(?, title),
          body   = COALESCE(?, body)
      WHERE id = ?`),
    del:    db.prepare('DELETE FROM posts WHERE id = ?'),
  },
  comments: {
    all:    db.prepare('SELECT * FROM comments'),
    byId:   db.prepare('SELECT * FROM comments WHERE id = ?'),
    byPost: db.prepare('SELECT * FROM comments WHERE postId = ?'),
    insert: db.prepare('INSERT INTO comments (postId, name, email, body) VALUES (?, ?, ?, ?)'),
    del:    db.prepare('DELETE FROM comments WHERE id = ?'),
  },
};

Обратите внимание на COALESCE(?, name) в запросах update. Это реализация семантики PATCH: если клиент прислал null для поля, в базу уйдёт старое значение. Так можно обновлять только те поля, которые реально хочется изменить.

6. Вспомогательные функции

Две функции, которые используются во всех обработчиках:

sendJSON(res, status, data)

Сериализует объект в JSON, ставит правильные заголовки (Content-Type, Content-Length) и завершает ответ. 

function sendJSON(res, status, data) {
  const body = JSON.stringify(data);
  res.writeHead(status, {
    'Content-Type': 'application/json; charset=utf-8',
    'Content-Length': Buffer.byteLength(body),
  });
  res.end(body);
}

readBody(req)

Читает тело запроса как поток (chunk'ами), собирает строку, парсит JSON. Возвращает Promise, потому что события data и end асинхронны. 

function readBody(req) {
  return new Promise((resolve, reject) => {
    let data = '';
    req.on('data', (c) => (data += c));
    req.on('end', () => {
      if (!data) return resolve({});
      try { resolve(JSON.parse(data)); } catch (e) { reject(e); }
    });
    req.on('error', reject);
  });
}

7. Обработчики ресурсов

Каждый ресурс обслуживается своей асинхронной функцией. Внутри — проверка метода и пути, вызов подготовленного запроса и отправка JSON-ответа. Разберём один обработчик — handlePosts, остальные два (handleUsers, handleComments) устроены аналогично. 

async function handlePosts(req, res, parts, method, search) {
  const id = parts[1] ? Number(parts[1]) : null;
  const userId = search.get('userId');

  // GET /posts?userId=2
  if (method === 'GET' && !id && userId)
    return sendJSON(res, 200, stmt.posts.byUser.all(Number(userId)));

  // GET /posts
  if (method === 'GET' && !id)
    return sendJSON(res, 200, stmt.posts.all.all());

  // GET /posts/:id
  if (method === 'GET' && id) {
    const p = stmt.posts.byId.get(id);
    return p ? sendJSON(res, 200, p) : sendJSON(res, 404, {});
  }

  // POST /posts
  if (method === 'POST') {
    const b = await readBody(req);
    if (!b.userId || !b.title || !b.body)
      return sendJSON(res, 400, { error: 'userId, title, body required' });
    const info = stmt.posts.insert.run(b.userId, b.title, b.body);
    return sendJSON(res, 201, stmt.posts.byId.get(info.lastInsertRowid));
  }

  // PUT /posts/:id и PATCH /posts/:id
  if ((method === 'PUT' || method === 'PATCH') && id) {
    const b = await readBody(req);
    const info = stmt.posts.update.run(
      b.userId ?? null, b.title ?? null, b.body ?? null, id
    );
    return info.changes
      ? sendJSON(res, 200, stmt.posts.byId.get(id))
      : sendJSON(res, 404, {});
  }

  // DELETE /posts/:id
  if (method === 'DELETE' && id) {
    const info = stmt.posts.del.run(id);
    return sendJSON(res, info.changes ? 200 : 404, {});
  }

  // всё остальное — 405
  sendJSON(res, 405, { error: 'Method not allowed' });
}

Ключевые детали:

  • parts[1] — второй сегмент URL. Для /posts/5 это "5", приводим к числу;
  • search.get('userId') — параметр из строки запроса (?userId=2);
  • info.lastInsertRowid — id только что созданной записи (для ответа 201);
  • info.changes — сколько строк реально изменилось. 0 значит «нет такой записи» → 404;
  • b.field ?? null — если поля нет, передаём null, чтобы COALESCE оставил старое значение.

8. HTTP-сервер и CORS

Корневой обработчик — createServer(async (req, res) => {...}). Он делает три вещи: ставит CORS-заголовки, разбирает URL на сегменты и передаёт управление нужному обработчику ресурса. 

const server = createServer(async (req, res) => {
  // CORS — чтобы фронтенд с другого домена мог обращаться к API
  res.setHeader('Access-Control-Allow-Origin', '*');
  res.setHeader('Access-Control-Allow-Methods', 'GET,POST,PUT,PATCH,DELETE,OPTIONS');
  res.setHeader('Access-Control-Allow-Headers', 'Content-Type');
  if (req.method === 'OPTIONS') { res.writeHead(204); return res.end(); }

  const url   = new URL(req.url, `http://${req.headers.host}`);
  const parts = url.pathname.split('/').filter(Boolean);
  const { method } = req;

  try {
    // Корень / — JSON-оглавление API
    if (parts.length === 0) {
      return sendJSON(res, 200, {
        name: 'SQLite Blog API',
        version: '1.0',
        endpoints: ['/users', '/posts', '/comments'],
      });
    }

    // Браузер сам дёргает /favicon.ico — тихо отвечаем 204
    if (parts[0] === 'favicon.ico') { res.writeHead(204); return res.end(); }

    // Маршрутизация по ресурсу
    if (parts[0] === 'users')    return await handleUsers(req, res, parts, method);
    if (parts[0] === 'posts')    return await handlePosts(req, res, parts, method, url.searchParams);
    if (parts[0] === 'comments') return await handleComments(req, res, parts, method, url.searchParams);

    sendJSON(res, 404, { error: 'Not found' });
  } catch (err) {
    console.error(err);
    sendJSON(res, 500, { error: err.message });
  }
});

server.listen(3000, '127.0.0.1', () => {
  console.log('API: http://127.0.0.1:3000');
});

url.pathname.split('/').filter(Boolean) превращает /posts/5 в ["posts", "5"], а / — в []. Это даёт простую и понятную маршрутизацию без сторонних библиотек.

9. Запуск

Проверьте версию Node.js — нужна 22.5 или выше:

node -v

Перейдите в папку со скриптом и запустите:

cd "D:\Мои документы\...\Зубарева Урванцева ИР-41"
node --no-warnings server.mjs

В консоли появится:

DB seeded
API: http://127.0.0.1:3000

Флаг --no-warnings скрывает безвредное сообщение ExperimentalWarning: SQLite is an experimental feature. Рядом со скриптом появится файл api.db — ваша база.

11. Примеры запросов

Получить список постов

curl.exe http://127.0.0.1:3000/posts

Создать пост

curl.exe -X POST http://127.0.0.1:3000/posts ^
  -H "Content-Type: application/json" ^
  -d "{"userId":1,"title":"Привет","body":"Мир"}"

Ответ (201 Created):

{
  "id": 101,
  "userId": 1,
  "title": "Привет",
  "body": "Мир"
}

Частично обновить пост

curl.exe -X PATCH http://127.0.0.1:3000/posts/101 ^
  -H "Content-Type: application/json" ^
  -d "{"title":"Обновлённый заголовок"}"

Удалить пост

curl.exe -X DELETE http://127.0.0.1:3000/posts/101

12. Клиентская разметка (markup)

Минимальная HTML-страница, которая запрашивает /posts у нашего API и выводит заголовки постов. Тот же Bootstrap, что и на этой странице. Достаточно сохранить файл и открыть в браузере (при запущенном server.mjs). 

<!DOCTYPE html>
<html lang="ru">
<head>
  <meta charset="UTF-8">
  <title>Посты из API</title>
  <link href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.0/dist/css/bootstrap.min.css" rel="stylesheet">
</head>
<body class="container py-4">
  <h1 class="mb-4">Посты</h1>
  <div id="posts" class="row g-3"></div>

  <script>
    async function loadPosts() {
      const res  = await fetch('http://127.0.0.1:3000/posts');
      const data = await res.json();

      const container = document.getElementById('posts');
      container.innerHTML = data.map(p => `
        <div class="col-md-6">
          <div class="card h-100 shadow-sm border-0">
            <div class="card-body">
              <h5 class="card-title fw-bold">${p.title}</h5>
              <p class="card-text text-muted">${p.body}</p>
              <span class="badge text-bg-secondary">userId: ${p.userId}</span>
            </div>
          </div>
        </div>
      `).join('');
    }

    loadPosts().catch(console.error);
  </script>
</body>
</html>

Запрос идёт через fetch, CORS-заголовки Access-Control-Allow-Origin: * в сервере разрешают обращение с любого домена — включая file:// и localhost. Если открыть эту разметку, увидите 100 карточек постов.
