Пример работы с SQLite на Python


import sqlite3

# 1. Подключение к базе данных (в оперативной памяти)
connection = sqlite3.connect(':memory:')
cursor = connection.cursor()

# 2. Создание таблиц
create_patients_query = '''
CREATE TABLE IF NOT EXISTS Patients (
    id INTEGER PRIMARY KEY,
    имя TEXT NOT NULL
);
'''

create_sessions_query = '''
CREATE TABLE IF NOT EXISTS Sessions (
    id INTEGER PRIMARY KEY,
    patient_id INTEGER,
    дата TEXT,
    FOREIGN KEY (patient_id) REFERENCES Patients(id)
);
'''

cursor.execute(create_patients_query)
cursor.execute(create_sessions_query)

# 3. Вставка одной записи в Patients
insert_patient_query = "INSERT INTO Patients (id, имя) VALUES (?, ?)"
patient_data = (1, 'Анна')
cursor.execute(insert_patient_query, patient_data)
connection.commit()

# 4. Вставка нескольких записей в Patients
patients_list = [
    (2, 'Борис'),
    (3, 'Виктор'),
    (4, 'Дарья'),
    (5, 'Елена'),
    (6, 'Сергей')
]
cursor.executemany(insert_patient_query, patients_list)
connection.commit()

# 5. Вставка одной записи в Sessions
insert_session_query = "INSERT INTO Sessions (id, patient_id, дата) VALUES (?, ?, ?)"
session_data = (1, 1, '2024-01-10')
cursor.execute(insert_session_query, session_data)
connection.commit()

# 6. Вставка нескольких записей в Sessions
sessions_list = [
    (2, 1, '2024-01-17'),
    (3, 3, '2024-01-15'),
    (4, 5, '2024-01-20'),
    (5, None, '2024-01-18'), # None превратится в NULL
    (6, 8, '2024-01-22')
]
cursor.executemany(insert_session_query, sessions_list)
connection.commit()

# 7. Выполнение основного SQL-запроса (поиск пациентов без сессий)
select_query = '''
SELECT Patients.id, Patients.имя 
FROM Patients 
LEFT JOIN Sessions ON Patients.id = Sessions.patient_id 
WHERE Sessions.patient_id IS NULL
'''
cursor.execute(select_query)
results = cursor.fetchall()

# 8. Вывод результатов
print("Пациенты, у которых нет сессий:")
for row in results:
    # row[0] - это id, row[1] - это имя
    print(f"ID: {row[0]}, Имя: {row[1]}")

# 9. Закрытие соединения
connection.close()
